# Состав локального синтеза Qwen3-TTS раздела «Презентации».
#
# Ставится в СВОЙ venv, отдельный от Silero, Piper и распознавания: torch с CUDA
# весит гигабайты, и его обновление не должно менять состав, на котором проверены
# остальные голоса.
#
# Колёса берутся с обычного PyPI, а не с индекса PyTorch: linux-сборки torch на
# PyPI уже несут CUDA (библиотеки nvidia-* приходят отдельными пакетами), а с
# `--index-url` PyTorch не нашлись бы остальные пакеты (transformers, librosa и
# соседи). Сам пакет qwen-tts torch не закрепляет — закрепляем здесь.
#
# Версии torch и torchaudio совпадают намеренно: у torchaudio есть собственные
# .so, собранные под torch той же версии, а самая свежая torchaudio на PyPI —
# 2.11.0 (2.13.0 не существует: проверено 11.09.2026 через pip index versions).
# flash-attn не ставится: по умолчанию используется sdpa.
#
# pysox закреплён на 1.4.1 — последней версии с готовым колесом: у 1.5.0 только
# sdist, чей setup.py импортирует numpy и падает в изолированной сборке pip.
# Сам qwen-tts версию sox не закрепляет.
qwen-tts==0.1.1
torch==2.11.0
torchaudio==2.11.0
sox==1.4.1
